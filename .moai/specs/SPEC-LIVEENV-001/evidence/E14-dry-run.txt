## 양성 대조군 ① — 좁은 판별자가 살아 있고, 증분 판정이 상수 함수가 아니다
# [HARD] 판정선을 증분으로 옮겼으므로 「증분이 생기면 실제로 빨개지는가」를 같은 파일에서 보인다.
#   대조군은 스스로 끝나는 프로세스로 세웠다 — 뒤에 kill 을 두면 그 줄에 닿지 못하는 경로가 남는다.

$ B=$(pgrep -f 'dangerously-load-development-channels' | wc -l)                       # → 1
$ python3 -c "import time; time.sleep(6)" --dangerously-load-development-channels probe-t35-sync &
$ D=$(pgrep -f 'dangerously-load-development-channels' | wc -l)                       # → 2  (대조군이 걸린다)
$ test "$B" -eq "$D"; echo "narrow-delta-zero exit=$?"
narrow-delta-zero exit=1        # 증분이 생기면 빨개진다 — 판정이 상수 함수가 아니다
$ wait   # 대조군이 스스로 끝난다
$ A=$(pgrep -f 'dangerously-load-development-channels' | wc -l)                       # → 1
$ test "$B" -eq "$A"; echo "narrow-delta-zero exit=$?"
narrow-delta-zero exit=0        # 되돌아온다
$ ps -p <대조군> > /dev/null; echo "대조군 종료 확인 exit=$?"
대조군 종료 확인 exit=1         # 죽었다
